#
# Testing
#

ENABLE_TESTING()

INCLUDE_DIRECTORIES(${CUDAToolkit_LIBRARY_DIR})

SET(OPENMM_BUILD_CUDA_DOUBLE_PRECISION_TESTS TRUE CACHE BOOL "Whether to build double precision versions of CUDA test cases")

SET( INCLUDE_SERIALIZATION FALSE )
#SET( INCLUDE_SERIALIZATION TRUE )

IF( INCLUDE_SERIALIZATION )
    INCLUDE_DIRECTORIES(${OPENMM_DIR}/serialization/include)
    SET( SHARED_OPENMM_SERIALIZATION "OpenMMSerialization" )
ENDIF( INCLUDE_SERIALIZATION )

# Automatically create tests using files named "Test*.cpp"
FILE(GLOB TEST_PROGS "*Test*.cpp")
FOREACH(TEST_PROG ${TEST_PROGS})
    GET_FILENAME_COMPONENT(TEST_ROOT ${TEST_PROG} NAME_WE)

    # Link with shared library
    ADD_EXECUTABLE(${TEST_ROOT} ${TEST_PROG})
    TARGET_LINK_LIBRARIES(${TEST_ROOT} ${SHARED_TARGET})
    IF (APPLE)
        SET_TARGET_PROPERTIES(${TEST_ROOT} PROPERTIES LINK_FLAGS "${EXTRA_COMPILE_FLAGS} -F/Library/Frameworks -framework CUDA" COMPILE_FLAGS "${EXTRA_COMPILE_FLAGS}")
    ELSE (APPLE)
        SET_TARGET_PROPERTIES(${TEST_ROOT} PROPERTIES LINK_FLAGS "${EXTRA_LINK_FLAGS}" COMPILE_FLAGS "${EXTRA_COMPILE_FLAGS}")
    ENDIF (APPLE)

    ADD_TEST(${TEST_ROOT}Single ${EXECUTABLE_OUTPUT_PATH}/${TEST_ROOT} single)
    IF (OPENMM_BUILD_CUDA_DOUBLE_PRECISION_TESTS)
        ADD_TEST(${TEST_ROOT}Mixed ${EXECUTABLE_OUTPUT_PATH}/${TEST_ROOT} mixed)
        ADD_TEST(${TEST_ROOT}Double ${EXECUTABLE_OUTPUT_PATH}/${TEST_ROOT} double)
    ENDIF(OPENMM_BUILD_CUDA_DOUBLE_PRECISION_TESTS)

ENDFOREACH(TEST_PROG ${TEST_PROGS})

# Independently registered spatial-data consumer, loaded by validate.py.
ADD_LIBRARY(SpatialWorkViewTestCUDA SHARED ${CMAKE_SOURCE_DIR}/platforms/common/tests/TestSpatialWorkView.cpp)
TARGET_INCLUDE_DIRECTORIES(SpatialWorkViewTestCUDA PRIVATE ${CMAKE_SOURCE_DIR}/platforms/common/tests)
TARGET_LINK_LIBRARIES(SpatialWorkViewTestCUDA PRIVATE ${SHARED_TARGET} OpenMM)
SET_TARGET_PROPERTIES(SpatialWorkViewTestCUDA PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR} RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
TARGET_COMPILE_DEFINITIONS(SpatialWorkViewTestCUDA PRIVATE DIAGNOSTIC_CUDA)

ADD_EXECUTABLE(TestSpatialWorkViewCUDA ${CMAKE_SOURCE_DIR}/platforms/common/tests/TestSpatialWorkView.cpp)
TARGET_INCLUDE_DIRECTORIES(TestSpatialWorkViewCUDA PRIVATE ${CMAKE_SOURCE_DIR}/platforms/common/tests)
TARGET_COMPILE_DEFINITIONS(TestSpatialWorkViewCUDA PRIVATE SPATIAL_TEST_EXECUTABLE DIAGNOSTIC_CUDA)
TARGET_LINK_LIBRARIES(TestSpatialWorkViewCUDA PRIVATE ${SHARED_TARGET} OpenMM)
ADD_TEST(NAME SpatialWorkViewCUDAMixed COMMAND TestSpatialWorkViewCUDA mixed)
ADD_TEST(NAME SpatialWorkViewCUDADouble COMMAND TestSpatialWorkViewCUDA double)
